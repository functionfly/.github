# Function Registry + Live Runtime Graph (FRG) Implementation Plan

## Executive Summary

This plan implements the FRG system by **extending existing capabilities** rather than building from scratch. The codebase already has:
- Function Registry (full versioning, trust scores, DRE)
- Agent Graph service (Node/Edge/Graph models)
- Sandbox execution engine (daemon mode, WASM)
- State management with triggers
- Deterministic execution (MEG, FXCERT)

**Strategy: Build the "Live Function Graph Engine" on top of these foundations.**

// CreateGraph: Build graph from nodes
func (h *Handler) CreateGraph(w http.ResponseWriter, r *http.Request) {
    var req CreateGraphRequest
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        respondError(w, http.StatusBadRequest, err)
        return
    }
    
    // Validate all function refs exist (reuse registry lookup)
    for _, nodeRef := range req.Nodes {
        fn, err := h.registryRepo.GetFunctionByID(r.Context(), nodeRef.FunctionID)
        if err != nil {
            respondError(w, http.StatusBadRequest, fmt.Errorf("function not found: %s", nodeRef.FunctionID))
            return
        }
        // Check permissions (reuse existing)
        if fn.Visibility != "public" && !h.hasAccess(r.Context(), fn) {
            respondError(w, http.StatusForbidden, fmt.Errorf("no access to function: %s", nodeRef.FunctionID))
            return
        }
    }
    
    // Detect cycles (reuse agent/graph service Kahn algorithm)
    if err := h.detectCycle(req.Edges); err != nil {
        respondError(w, http.StatusBadRequest, err)
        return
    }
    
    // Create graph definition
    graph, err := h.graphRepo.Create(r.Context(), req.toModel())
    if err != nil {
        respondError(w, http.StatusInternalServerError, err)
        return
    }
    
    respondJSON(w, http.StatusCreated, graph)
}

// ExecuteGraph: Run graph once
func (h *Handler) ExecuteGraph(w http.ResponseWriter, r *http.Request) {
    vars := mux.Vars(r)
    author, name := vars["author"], vars["name"]
    version := r.URL.Query().Get("version")
    if version == "" {
        version = "latest"
    }
    
    // Get graph definition
    graph, err := h.graphRepo.GetByName(r.Context(), author, name, version)
    if err != nil {
        respondError(w, http.StatusNotFound, err)
        return
    }
    
    // Parse input
    var input map[string]interface{}
    json.NewDecoder(r.Body).Decode(&input)
    
    // Execute (async or sync based on graph.execution_mode)
    if graph.ExecutionMode == "sync" {
        result, err := h.engine.ExecuteSync(r.Context(), graph, input)
        if err != nil {
            respondError(w, http.StatusInternalServerError, err)
            return
        }
        respondJSON(w, http.StatusOK, result)
    } else {
        // Async: return execution ID, background execution
        instance, err := h.engine.CreateInstance(r.Context(), graph, input)
        if err != nil {
            respondError(w, http.StatusInternalServerError, err)
            return
        }
        respondJSON(w, http.StatusAccepted, map[string]string{
            "execution_id": instance.ID.String(),
            "status": "pending",
        })
    }
}

// StreamGraph: WebSocket or SSE streaming
func (h *Handler) StreamGraph(w http.ResponseWriter, r *http.Request) {
    // Upgrade to WebSocket or use SSE based on client preference
    if r.Header.Get("Upgrade") == "websocket" {
        h.streamWebSocket(w, r)
    } else {
        h.streamSSE(w, r)
    }
}

package frg

import (
    "context"
    "sync"
    
    "github.com/functionfly/internal/api/handlers/registry/execution"
    "github.com/functionfly/internal/dre"
    "github.com/functionfly/internal/storage"
)

// ExecutionEngine orchestrates graph execution
type ExecutionEngine struct {
    // Reuse existing components
    executorPool    *execution.SandboxClient  // daemon mode executor
    registryRepo    *storage.RegistryRepository
    stateService    *storage.StateService     // for node state
    
    // Live instances
    instances       map[uuid.UUID]*GraphInstance
    instancesMu     sync.RWMutex
    
    // Event bus (reuse existing pub/sub)
    eventBus        EventBus
    
    // DRE (existing)
    megBuilder      *dre.MEGBuilder
    
    // Streaming
    streamManager   *StreamManager
}

// ExecuteSync: Synchronous DAG execution
func (e *ExecutionEngine) ExecuteSync(ctx context.Context, graph *GraphDefinition, input map[string]interface{}) (*GraphResult, error) {
    // 1. Create ephemeral instance
    instance, err := e.createInstance(ctx, graph, input)
    if err != nil {
        return nil, err
    }
    
    // 2. Compute execution order (reuse Kahn from agent/graph)
    order, err := e.getExecutionOrder(instance.Nodes, instance.Edges)
    if err != nil {
        return nil, err
    }
    
    // 3. Execute nodes in topological order
    for _, nodeID := range order {
        node := instance.Nodes[nodeID]
        
        // Collect inputs from predecessors
        nodeInput := e.collectInputs(node, instance.Edges)
        
        // Execute via existing sandbox
        result, err := e.executeNode(ctx, node, nodeInput)
        if err != nil {
            // Check for fallback
            if node.FallbackNodeID != nil {
                fallback := instance.Nodes[*node.FallbackNodeID]
                result, err = e.executeNode(ctx, fallback, nodeInput)
            }
            if err != nil {
                instance.Status = GraphStatusFailed
                return nil, e.buildGraphResult(instance, err)
            }
        }
        
        node.Status = NodeStatusCompleted
        node.OutputData = result.Data
        node.ExecutionCert = result.Cert
    }
    
    // 4. Build MEG (reuse existing DRE)
    instance.ExecutionRootHash = e.megBuilder.BuildMEG(instance).ExecutionRootHash
    
    // 5. Return result
    return e.buildGraphResult(instance, nil), nil
}

// ExecuteAsync: Create persistent streaming instance
func (e *ExecutionEngine) ExecuteAsync(ctx context.Context, graph *GraphDefinition, input map[string]interface{}) (*GraphInstance, error) {
    instance, err := e.createInstance(ctx, graph, input)
    if err != nil {
        return nil, err
    }
    
    // Set up streaming channels
    instance.InputChannel = make(chan *ExecutionEvent, 100)
    instance.OutputChannel = make(chan *ExecutionResult, 100)
    
    // Store instance
    e.instancesMu.Lock()
    e.instances[instance.ID] = instance
    e.instancesMu.Unlock()
    
    // Start runtime loop
    go e.runInstance(instance)
    
    return instance, nil
}

// runInstance: Live event loop for streaming/event-driven graphs
func (e *ExecutionEngine) runInstance(instance *GraphInstance) {
    instance.Status = GraphStatusStreaming
    instance.StartedAt = timePtr(time.Now())
    
    // Event-driven execution
    for event := range instance.InputChannel {
        switch event.Type {
        case EventTypeTrigger:
            e.handleTrigger(instance, event)
        case EventTypeComplete:
            e.handleNodeComplete(instance, event)
        case EventTypeError:
            e.handleNodeError(instance, event)
        case EventTypeStream:
            e.handleStreamEvent(instance, event)
        case EventTypeCheckpoint:
            e.handleCheckpoint(instance, event)
        }
    }
    
    instance.Status = GraphStatusCompleted
    instance.CompletedAt = timePtr(time.Now())
}

func (e *ExecutionEngine) executeNode(ctx context.Context, node *NodeInstance, input interface{}) (*NodeResult, error) {
    node.Status = NodeStatusExecuting
    node.AttemptCount++
    node.LastAttemptAt = timePtr(time.Now())
    
    // Use existing sandbox executor (daemon mode for low latency)
    execRequest := &execution.ExecuteRequest{
        FunctionID:    node.NodeRef.FunctionID.String(),
        Version:       node.NodeRef.Version,
        Input:         input,
        TimeoutMs:     node.NodeRef.TimeoutMs,
        MemoryMB:      node.NodeRef.MemoryMB,
        Deterministic: node.NodeRef.Deterministic,
    }
    
    result, err := e.executorPool.Execute(ctx, execRequest)
    if err != nil {
        return nil, err
    }
    
    return &NodeResult{
        Data:       result.Output,
        Cert:       result.Certificate,
        DurationMs: result.DurationMs,
    }, nil
}

// executeNodeParallel: Auto-parallelization for independent nodes
func (e *ExecutionEngine) executeNodesParallel(ctx context.Context, nodes []*NodeInstance, inputs map[string]interface{}) ([]*NodeResult, error) {
    var wg sync.WaitGroup
    errChan := make(chan error, len(nodes))
    resultChan := make(chan *NodeResult, len(nodes))
    
    for _, node := range nodes {
        wg.Add(1)
        go func(n *NodeInstance) {
            defer wg.Done()
            result, err := e.executeNode(ctx, n, inputs[n.ID])
            if err != nil {
                errChan <- err
                return
            }
            resultChan <- result
        }(node)
    }
    
    wg.Wait()
    close(errChan)
    close(resultChan)
    
    // Check for errors
    for err := range errChan {
        if err != nil {
            return nil, err
        }
    }
    
    // Collect results
    var results []*NodeResult
    for r := range resultChan {
        results = append(results, r)
    }
    
    return results, nil
}

package frg

// StreamManager handles streaming edges

type StreamManager struct {
    streams map[string]*StreamHandler
    mu      sync.RWMutex
}

type StreamHandler struct {
    EdgeID       uuid.UUID
    SourceNode   *NodeInstance
    TargetNode   *NodeInstance
    
    // Ring buffer for backpressure
    Buffer       *RingBuffer
    
    // Transform pipeline
    Transformer  DataTransformer
    
    // Windowing for aggregations
    Window       *TimeWindow
}

// Handle continuous data flow between streaming nodes
func (sm *StreamManager) ProcessStream(ctx context.Context, handler *StreamHandler, event *ExecutionEvent) {
    switch handler.Transformer.Type {
    case TransformMap:
        mapped := handler.Transformer.Map(event.Data)
        handler.TargetNode.InputChannel <- mapped
        
    case TransformWindow:
        handler.Window.Add(event.Data)
        if handler.Window.IsReady() {
            aggregated := handler.Window.Aggregate()
            handler.TargetNode.InputChannel <- aggregated
        }
        
    case TransformFilter:
        if handler.Transformer.Filter(event.Data) {
            handler.TargetNode.InputChannel <- event.Data
        }
    }
}

// AnalyzeGraph: Continuous optimization

            Action:      Action{Type: "enable_cache", NodeID: "Z"},
        })
    }
    
    // Suggest faster alternatives
    alternatives := o.findFasterAlternatives(graphID, metrics)
    for _, alt := range alternatives {
        suggestions = append(suggestions, &OptimizationSuggestion{
            Type:        "replacement",
            Description: fmt.Sprintf("Replace %s with %s (%.1fx faster)", alt.Current, alt.Alternative, alt.Speedup),
            Impact:      1.0 - (1.0/alt.Speedup),
            Action:      Action{Type: "replace", From: alt.Current, To: alt.Alternative},
        })
    }
    
    return suggestions, nil
}

func (e *ExecutionEngine) calculateGraphCost(instance *GraphInstance) *CostBreakdown {
    total := 0.0
    breakdown := &CostBreakdown{
        NodeCosts: make(map[string]float64),
    }
    
    for _, node := range instance.Nodes {
        // Get function price from registry
        price := node.NodeRef.PricePerCall
        
        // Add runtime cost (CPU time, memory)
        runtimeCost := e.calculateRuntimeCost(node)
        
        nodeCost := price + runtimeCost
        breakdown.NodeCosts[node.ID] = nodeCost
        total += nodeCost
    }
    
    breakdown.Total = total
    breakdown.PlatformFee = total * 0.15 // 15% platform fee
    breakdown.CreatorEarnings = total - breakdown.PlatformFee
    
    return breakdown
}

// LocalRuntime for development

type LocalRuntime struct {
    engine      *frg.ExecutionEngine
    fileWatcher *fsnotify.Watcher
    hotReload   bool
}

// Watch and auto-reload graph during development
func (r *LocalRuntime) WatchGraph(path string) error {
    // Load graph definition from local file
    graph, err := r.loadFromFile(path)
    if err != nil {
        return err
    }
    
    // Watch for changes
    go func() {
        for event := range r.fileWatcher.Events {
            if event.Op&fsnotify.Write == fsnotify.Write {
                // Hot reload
                newGraph, _ := r.loadFromFile(path)
                r.engine.UpdateGraph(newGraph)
            }
        }
    }()
    
    // Start local server
    return r.serveLocal(graph)
}
