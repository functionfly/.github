# Team Memory Engine Implementation Plan

## Overview
Build a "Shared Brain" - a central team memory layer that stores decisions, preferences, processes, and client context. Accessible by all agents/tools with auto-update capabilities from conversations.

---

## Architecture Foundation

The FunctionFly platform already provides excellent primitives:
- **AgentMemory** model with pgvector embeddings (1536-dim)
- **State Fabric** with event sourcing and versioning
- **Team/Organization** structure with role-based access
- **pgvector + HNSW** indexes for semantic search
- **Multi-tenancy** via RLS policies
- **TTL/Retention** with ImportanceScore and ExpiresAt

---

```go
    ListByType(ctx context.Context, tenantID, teamID uuid.UUID, memoryType string) ([]*models.TeamMemory, error)
    ListByCategory(ctx context.Context, tenantID, teamID uuid.UUID, category string) ([]*models.TeamMemory, error)
    
    // Semantic search (vector similarity)
    SearchSimilar(ctx context.Context, tenantID, teamID uuid.UUID, queryVector []float32, limit int) ([]*models.TeamMemory, error)
    SearchByText(ctx context.Context, tenantID, teamID uuid.UUID, searchText string, limit int) ([]*models.TeamMemory, error)
    
    // Auto-update related
    FindByConversation(ctx context.Context, conversationID uuid.UUID) ([]*models.TeamMemory, error)
    MarkAsAccessed(ctx context.Context, memoryID uuid.UUID) error
    UpdateFromConversation(ctx context.Context, memoryID uuid.UUID, newContent models.JSONMap, confidence float64) error
    
    // Maintenance
    CleanupExpired(ctx context.Context, batchSize int) (int, error)
    ListForAutoUpdate(ctx context.Context, batchSize int) ([]*models.TeamMemory, error)
}

type MemoryFilter struct {
    MemoryType   *string
    Category     *string
    IsValidated  *bool
    MinConfidence *float64
    CreatedAfter  *time.Time
    Limit         int
    Offset        int
}

type teamMemoryRepo struct {
    db *gorm.DB
    embeddingService EmbeddingService // For text-to-vector conversion
}

func NewTeamMemoryRepository(db *gorm.DB, embedSvc EmbeddingService) TeamMemoryRepository {
    return &teamMemoryRepo{db: db, embeddingService: embedSvc}
}

// SearchSimilar performs vector similarity search using pgvector
func (r *teamMemoryRepo) SearchSimilar(ctx context.Context, tenantID, teamID uuid.UUID, 
    queryVector []float32, limit int) ([]*models.TeamMemory, error) {
    
    var memories []*models.TeamMemory
    vectorStr := pgVectorToString(queryVector)
    
    err := r.db.WithContext(ctx).Raw(`
        SELECT * FROM team_memories
        WHERE tenant_id = ? AND team_id = ?
          AND (expires_at IS NULL OR expires_at > NOW())
        ORDER BY embedding <=> ?
        LIMIT ?`, tenantID, teamID, vectorStr, limit).Scan(&memories).Error
    
    return memories, err
}

// SearchByText converts text to embedding then searches
func (r *teamMemoryRepo) SearchByText(ctx context.Context, tenantID, teamID uuid.UUID, 
    searchText string, limit int) ([]*models.TeamMemory, error) {
    
    // Generate embedding for search text
    embedding, err := r.embeddingService.GenerateEmbedding(ctx, searchText)
    if err != nil {
        return nil, err
    }
    
    return r.SearchSimilar(ctx, tenantID, teamID, embedding, limit)
}

// MarkAsAccessed updates access tracking
func (r *teamMemoryRepo) MarkAsAccessed(ctx context.Context, memoryID uuid.UUID) error {
    return r.db.WithContext(ctx).Model(&models.TeamMemory{}).
        Where("id = ?", memoryID).
        Updates(map[string]interface{}{
            "access_count": gorm.Expr("access_count + 1"),
            "last_accessed_at": time.Now(),
        }).Error
}
```

```go
func (h *Handler) HandleCreateMemory(w http.ResponseWriter, r *http.Request) {
    user := middleware.GetUserFromContext(r)
    vars := mux.Vars(r)
    teamID, _ := uuid.Parse(vars["teamId"])
    
    // Verify team membership
    if !h.isTeamMember(user.UserID, teamID) {
        http.Error(w, "Access denied", http.StatusForbidden)
        return
    }
    
    var req CreateMemoryRequest
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, err.Error(), http.StatusBadRequest)
        return
    }
    
    // Generate embedding from summary + content
    embeddingText := req.Summary + " " + req.Content.ToSearchableText()
    embedding, err := h.embeddingService.GenerateEmbedding(r.Context(), embeddingText)
    
    // Create memory with embedding
    memory := &models.TeamMemory{
        TenantID:   user.TenantID,
        TeamID:     teamID,
        MemoryType: req.MemoryType,
        Category:   req.Category,
        Content:    req.Content,
        Summary:    &req.Summary,
        CreatedBy:  user.UserID,
        Embedding:  embedding,
        ConfidenceScore: req.Confidence,
    }
    
    result, err := h.repo.Create(r.Context(), memory)
    json.NewEncoder(w).Encode(result)
}
```

```go
func (h *Handler) HandleSearchMemories(w http.ResponseWriter, r *http.Request) {
    user := middleware.GetUserFromContext(r)
    vars := mux.Vars(r)
    teamID, _ := uuid.Parse(vars["teamId"])
    
    var req SearchRequest
    json.NewDecoder(r.Body).Decode(&req)
    
    // Option A: Text search (auto-generate embedding)
    if req.Query != "" {
        memories, err := h.repo.SearchByText(r.Context(), user.TenantID, teamID, 
            req.Query, req.Limit)
        // Track access for each result
        for _, m := range memories {
            h.repo.MarkAsAccessed(r.Context(), m.ID)
        }
        json.NewEncoder(w).Encode(SearchResponse{Memories: memories})
        return
    }
    
    // Option B: Vector search (direct embedding provided)
    if len(req.Embedding) > 0 {
        memories, err := h.repo.SearchSimilar(r.Context(), user.TenantID, teamID,
            req.Embedding, req.Limit)
        json.NewEncoder(w).Encode(SearchResponse{Memories: memories})
    }
}
```

```go
package team_memory

import (
    "context"

    "strings"
    "functionfly/internal/storage"
    "functionfly/internal/storage/models"
)

type AutoUpdater struct {
    memoryRepo storage.TeamMemoryRepository
    convRepo   storage.ConversationRepository
    aiClient   AIAnalysisClient // Interface to AI service for extraction
}

// MemoryExtractionResult from AI analysis
type MemoryExtractionResult struct {
    Memories []PotentialMemory `json:"memories"`
    Confidence float64          `json:"confidence"`
}

type PotentialMemory struct {
    Type        string          `json:"type"` // decision, preference, process, client_context
    Category    string          `json:"category,omitempty"`
    Content     interface{}     `json:"content"`
    Summary     string          `json:"summary"`
    Confidence  float64         `json:"confidence"`
    Rationale   string          `json:"rationale"`
}

// ProcessConversation analyzes a completed conversation for memory-worthy content
func (u *AutoUpdater) ProcessConversation(ctx context.Context, conversationID uuid.UUID) error {
    // 1. Fetch conversation with all messages
    conv, err := u.convRepo.GetByID(ctx, conversationID)
    
    // 2. Build conversation transcript
    transcript := buildTranscript(conv.Messages)
    
    // 3. Call AI service to extract potential memories
    prompt := buildExtractionPrompt(transcript)
    result, err := u.aiClient.AnalyzeConversation(ctx, prompt)
    
    // 4. For each potential memory, check if it conflicts with existing
    for _, potential := range result.Memories {
        existing, err := u.findSimilarMemory(ctx, conv.TeamID, potential)
        
        if existing != nil {
            // Merge or update existing memory
            u.mergeOrUpdateMemory(ctx, existing, potential)
        } else if potential.Confidence >= 0.8 {
            // Create new memory (high confidence only)
            u.createMemoryFromExtraction(ctx, conv, potential)
        }
    }
    
    return nil
}
```

```go
// AgentContextProvider injects team memory into agent context
type AgentContextProvider struct {
    memoryRepo storage.TeamMemoryRepository
}

// BuildContextForAgent generates context string for LLM/agent
func (p *AgentContextProvider) BuildContextForAgent(
    ctx context.Context,
    tenantID, teamID uuid.UUID,
    currentTask string, // What the agent is currently working on
    relevantCategories []string, // e.g., ["client:acme", "process:onboarding"]
) (string, error) {
    
    // 1. Generate embedding from current task
    taskEmbedding, _ := p.generateEmbedding(ctx, currentTask)
    
    // 2. Semantic search for relevant memories
    memories, _ := p.memoryRepo.SearchSimilar(ctx, tenantID, teamID, taskEmbedding, 10)
    
    // 3. Also get recent validated decisions
    recentDecisions, _ := p.memoryRepo.ListByType(ctx, tenantID, teamID, "decision")
    
    // 4. Build formatted context block
    var contextParts []string
    
    if len(memories) > 0 {
        contextParts = append(contextParts, "## Team Knowledge & Context")
        for _, m := range memories {
            contextParts = append(contextParts, formatMemoryForAgent(m))
        }
    }
    
    return strings.Join(contextParts, "\n\n"), nil
}

func formatMemoryForAgent(m *models.TeamMemory) string {
    validity := ""
    if !m.IsValidated {
        validity = " [UNVALIDATED - use with caution]"
    }
    
    return fmt.Sprintf(`### %s (%s)%s
%s
Confidence: %.0f%% | Last updated: %s`,
        m.Summary,
        m.MemoryType,
        validity,
        formatContent(m.Content),
        m.ConfidenceScore * 100,
        m.UpdatedAt.Format("2006-01-02"),
    )
}
```

#### 6.4 Client-Side Encryption Toggle (User-Controlled)

**Goal:** Allow users to enable/disable encryption per memory (zero-knowledge for sensitive data).

**Storage Model:**
- `is_encrypted: BOOLEAN DEFAULT false` - user toggle
- `encrypted_content: BYTEA` - AES-256-GCM ciphertext
- `encryption_iv: BYTEA` - 12-byte nonce
- `encryption_tag: BYTEA` - 16-byte auth tag

**Client-Side Crypto Module:**

```typescript
import { createHash } from 'crypto-browserify';

export class TeamMemoryCrypto {
  private key: CryptoKey;
  
  // Derive key from team passphrase using PBKDF2
  static async deriveKey(passphrase: string, salt: Uint8Array): Promise<CryptoKey> {
    const encoder = new TextEncoder();
    const keyMaterial = await crypto.subtle.importKey(
      'raw', encoder.encode(passphrase), 'PBKDF2', false, ['deriveKey']
    );
    return crypto.subtle.deriveKey(
      { name: 'PBKDF2', salt, iterations: 100000, hash: 'SHA-256' },
      keyMaterial,
      { name: 'AES-GCM', length: 256 },
      false,
      ['encrypt', 'decrypt']
    );
  }
  
  // Encrypt memory content before sending to API
  async encrypt(content: object): Promise<{ ciphertext: Uint8Array, iv: Uint8Array, tag: Uint8Array }> {
    const iv = crypto.getRandomValues(new Uint8Array(12));
    const encoder = new TextEncoder();
    const plaintext = encoder.encode(JSON.stringify(content));
    
    const encrypted = await crypto.subtle.encrypt(
      { name: 'AES-GCM', iv },
      this.key,
      plaintext
    );
    
    // Split ciphertext and auth tag (last 16 bytes)
    const ciphertext = new Uint8Array(encrypted.slice(0, -16));
    const tag = new Uint8Array(encrypted.slice(-16));
    
    return { ciphertext, iv, tag };
  }
  
  // Decrypt after receiving from API
  async decrypt(ciphertext: Uint8Array, iv: Uint8Array, tag: Uint8Array): Promise<object> {
    // Concatenate ciphertext + tag for WebCrypto
    const combined = new Uint8Array(ciphertext.length + tag.length);
    combined.set(ciphertext);
    combined.set(tag, ciphertext.length);
    
    const decrypted = await crypto.subtle.decrypt(
      { name: 'AES-GCM', iv },
      this.key,
      combined
    );
    
    const decoder = new TextDecoder();
    return JSON.parse(decoder.decode(decrypted));
  }
}

// Team passphrase stored in sessionStorage (not localStorage - more secure)
// Prompt user for passphrase when accessing encrypted memories
```

**API Flow:**

1. **Create Memory (Encrypted):**
   ```
   Client: Encrypt content → send {is_encrypted: true, encrypted_content, iv, tag}
   API: Store only, no server-side decryption
   ```

2. **Search (Always Works):**
   - Search uses embedding (generated from plaintext before encryption)
   - Server can find relevant encrypted memories
   - Client decrypts after retrieval

3. **Retrieval:**
   ```
   API: Return {is_encrypted: true, encrypted_content, iv, tag, summary, embedding}
   Client: Decrypt with team passphrase
   ```

**UI Toggle:**
- Checkbox "Encrypt this memory" (default: off)
- Warns: "Encrypted memories require team passphrase to view"
- For auto-extracted memories: respect team default setting

**Security Benefits:**
- Server never sees plaintext of encrypted memories
- Database breaches don't expose sensitive client data
- Compliance: GDPR/CCPA data protection by default
- Trade-off: Cannot search within encrypted content JSON (only embedding-based search)
